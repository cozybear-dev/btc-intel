# btc-intel Plan 1 — Foundation + Knowledge Base (Phase 0) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create the repository skeleton and the research knowledge base. The knowledge base has two parts: a machine-readable catalog of analysis heuristics, obfuscation/laundering techniques and incident case studies, and cited Markdown pages. A checker enforces the catalog's integrity.

**Architecture:** The `catalog/*.yaml` files are the single source of truth, validated by pydantic models (`btcintel.catalog.schema`). A cross-checker (`btcintel.catalog.loader.check_catalog`) enforces:
- unique IDs
- references that resolve
- consistent heuristic/technique/incident links
- valid Bitcoin seeds
- a cited KB page for every entry

`docs/kb/index.md` and `docs/kb/ttp-matrix.md` are generated from the catalog. Plan 2's detectors register against the same heuristic IDs.

**Tech Stack:** Python 3.13, uv, pydantic v2, PyYAML, pytest, ruff, and the stdlib `urllib` for the network checks.

**Spec:** `docs/superpowers/specs/2026-09-25-btc-intel-design.md`. Read its "Research summary" section; it is the content outline for Tasks 5–7.

**Plan series:** Plan 1 (this document) → Plan 2 `2026-09-25-btc-intel-2-engine-api.md` → Plan 3 `2026-09-25-btc-intel-3-workbench.md`.

## Review Focus

1. **A catalog file containing a mapping (or a typo that makes YAML a string) instead of a list.** The user expects a one-line error naming the file, not a pydantic traceback. Test: Task 3, `test_list_file_must_be_a_list`.

3. **Seed values with stray whitespace or an uppercase bech32 address copied from a PDF.** Whitespace must be reported. Uppercase bech32 is valid and its type is recognised. Tests: Task 2, `test_uppercase_bech32_is_valid`; Task 3, `test_seed_with_whitespace_reported`.

5. **Generated docs drifting from the catalog, or reordering between runs.** Rendering must be deterministic and the committed files must match. Tests: Task 8, `test_render_is_deterministic` and `test_generated_docs_up_to_date`.

```python
"""Local validation of Bitcoin mainnet addresses, txids and search queries.

No network access: base58check (P2PKH/P2SH) and bech32/bech32m (BIP173/BIP350) checksums.
"""

from __future__ import annotations

import hashlib
import re
from typing import Literal

AddressType = Literal["p2pkh", "p2sh", "v0_p2wpkh", "v0_p2wsh", "v1_p2tr", "witness_unknown"]
QueryKind = Literal["address", "txid", "outpoint", "text"]

_B58 = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz"
_BECH32 = "qpzry9x8gf2tvdw0s3jn54khce6mua7l"
_BECH32M_CONST = 0x2BC830A3
_TXID_RE = re.compile(r"^[0-9a-fA-F]{64}$")


def _b58decode_check(s: str) -> bytes | None:
    n = 0
    for c in s:
        i = _B58.find(c)
        if i < 0:
            return None
        n = n * 58 + i
    body = n.to_bytes((n.bit_length() + 7) // 8, "big") if n else b""
    raw = b"\x00" * (len(s) - len(s.lstrip("1"))) + body
    if len(raw) < 5:
        return None
    payload, checksum = raw[:-4], raw[-4:]
    if hashlib.sha256(hashlib.sha256(payload).digest()).digest()[:4] != checksum:
        return None
    return payload


def _polymod(values: list[int]) -> int:
    gen = (0x3B6A57B2, 0x26508E6D, 0x1EA119FA, 0x3D4233DD, 0x2A1462B3)
    chk = 1
    for v in values:
        top = chk >> 25
        chk = (chk & 0x1FFFFFF) << 5 ^ v
        for i in range(5):
            if (top >> i) & 1:
                chk ^= gen[i]
    return chk


def _hrp_expand(hrp: str) -> list[int]:
    return [ord(x) >> 5 for x in hrp] + [0] + [ord(x) & 31 for x in hrp]


def _convertbits(data: list[int], frombits: int, tobits: int) -> list[int] | None:
    """Regroup bits without padding (decoding direction); None if padding is invalid."""
    acc = bits = 0
    ret: list[int] = []
    maxv = (1 << tobits) - 1
    for value in data:
        acc = (acc << frombits) | value
        bits += frombits
        while bits >= tobits:
            bits -= tobits
            ret.append((acc >> bits) & maxv)
    if bits >= frombits or ((acc << (tobits - bits)) & maxv):
        return None
    return ret


def _segwit_decode(addr: str) -> tuple[int, bytes] | None:
    if addr.lower() != addr and addr.upper() != addr:
        return None
    addr = addr.lower()
    pos = addr.rfind("1")
    if addr[:pos] != "bc" or pos + 7 > len(addr) or len(addr) > 90:
        return None
    data = [_BECH32.find(c) for c in addr[pos + 1 :]]
    if any(d < 0 for d in data):
        return None
    const = _polymod(_hrp_expand("bc") + data)
    if const not in (1, _BECH32M_CONST) or not data[:-6]:
        return None
    witver, prog5 = data[0], data[1:-6]
    prog = _convertbits(prog5, 5, 8)
    if prog is None or not 2 <= len(prog) <= 40 or witver > 16:
        return None
    if witver == 0 and (len(prog) not in (20, 32) or const != 1):
        return None
    if witver != 0 and const != _BECH32M_CONST:
        return None
    return witver, bytes(prog)


def address_type(addr: str) -> AddressType | None:
    """Return the script type of a valid mainnet address, or None if invalid."""
    if not addr or not addr.isascii():
        return None
    if addr[:3].lower() == "bc1":
        decoded = _segwit_decode(addr)
        if decoded is None:
            return None
        witver, prog = decoded
        if witver == 0:
            return "v0_p2wpkh" if len(prog) == 20 else "v0_p2wsh"
        if witver == 1 and len(prog) == 32:
            return "v1_p2tr"
        return "witness_unknown"
    if addr[0] in "13" and 25 <= len(addr) <= 35:
        payload = _b58decode_check(addr)
        if payload is None or len(payload) != 21:
            return None
        return {0x00: "p2pkh", 0x05: "p2sh"}.get(payload[0])  # type: ignore[return-value]
    return None


def is_address(s: str) -> bool:
    return address_type(s) is not None


def normalize_address(addr: str) -> str:
    """Bech32 addresses are case-insensitive; store them lowercase."""
    return addr.lower() if addr[:3].lower() == "bc1" else addr


def is_txid(s: str) -> bool:
    return bool(_TXID_RE.match(s))


def classify_query(q: str) -> tuple[QueryKind, str]:
    """Classify a search string and return it normalized."""
    q = q.strip()
    if is_txid(q):
        return "txid", q.lower()
    txid, sep, vout = q.partition(":")
    if sep and is_txid(txid) and vout.isdigit():
        return "outpoint", f"{txid.lower()}:{int(vout)}"
    if is_address(q):
        return "address", normalize_address(q)
    return "text", q
```

```python
"""Pydantic models for the machine-readable knowledge-base catalog (catalog/*.yaml)."""

from __future__ import annotations

from functools import cached_property
from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, model_validator

RiskLevel = Literal["none", "low", "medium", "high", "severe"]
RISK_ORDER: dict[str, int] = {"none": 0, "low": 1, "medium": 2, "high": 3, "severe": 4}

Tactic = Literal["acquisition", "layering", "obfuscation", "chain_hopping", "offchain", "integration"]
FatfSection = Literal[
    "transactions", "transaction_patterns", "anonymity", "senders_recipients",
    "source_of_funds", "geographical",
]


class _Entry(BaseModel):
    model_config = ConfigDict(extra="forbid", frozen=True)


class Reference(_Entry):
    id: str = Field(pattern=r"^[a-z0-9][a-z0-9_-]*$")
    title: str
    authors: list[str] = []
    year: int
    kind: Literal["paper", "court", "government", "industry", "docs", "news", "dataset", "code"]
    url: str = Field(pattern=r"^https?://")
    venue: str | None = None
    note: str | None = None


class Category(_Entry):
    id: str = Field(pattern=r"^[a-z][a-z0-9_]*$")
    kind: Literal["entity", "abuse"]
    name: str
    risk: RiskLevel
    gs_concepts: list[str] = []
    stop_trace: bool = False
    description: str


class Heuristic(_Entry):
    id: str = Field(pattern=r"^H-[A-Z0-9-]+$")
    name: str
    family: Literal[
        "clustering", "change", "fingerprint", "coinjoin", "pattern", "crosschain",
        "attribution", "taint", "temporal", "network",
    ]
    status: Literal["implemented", "planned", "kb_only"]
    implemented_by: str | None = None
    parent: str | None = None
    confidence: Literal["high", "medium", "low"]
    summary: str
    signals: list[str]
    false_positives: list[str] = []
    refs: list[str]
    kb: str

    @model_validator(mode="after")
    def _implemented_needs_target(self) -> Heuristic:
        if (self.status == "implemented") != (self.implemented_by is not None):
            raise ValueError("status 'implemented' requires implemented_by (and only then)")
        return self


class Technique(_Entry):
    id: str = Field(pattern=r"^O-[A-Z0-9-]+$")
    name: str
    tactic: Tactic
    summary: str
    examples: list[str] = []
    evades: list[str] = []
    detected_by: list[str] = []
    oak: list[str] = []
    fatf: list[FatfSection] = []
    refs: list[str]
    kb: str

    @model_validator(mode="after")
    def _oak_format(self) -> Technique:
        import re

        bad = [o for o in self.oak if not re.fullmatch(r"OAK-T\d+\.\d{3}", o)]
        if bad:
            raise ValueError(f"malformed OAK ids: {bad}")
        return self


class Seed(_Entry):
    kind: Literal["address", "tx"]
    value: str
    role: Literal["victim", "theft", "laundering", "exit", "seizure", "ransom", "other"]
    ref: str
    note: str | None = None


class Incident(_Entry):
    id: str = Field(pattern=r"^I-\d{4}-[A-Z0-9-]+$")
    name: str
    date: str = Field(pattern=r"^\d{4}(-\d{2}(-\d{2})?)?$")
    loss: str
    btc: float | None = None
    vector: list[str]
    actor: str | None = None
    attribution: Literal["confirmed", "suspected", "unknown"]
    techniques: list[str]
    summary: str
    seeds: list[Seed] = []
    seeds_note: str | None = None
    refs: list[str]
    kb: str

    @model_validator(mode="after")
    def _seeds_or_note(self) -> Incident:
        if not self.seeds and not self.seeds_note:
            raise ValueError("incident needs at least one seed or a seeds_note explaining why none")
        return self


class Catalog(_Entry):
    model_config = ConfigDict(extra="forbid", frozen=True, ignored_types=(cached_property,))

    references: list[Reference]
    categories: list[Category]
    heuristics: list[Heuristic]
    techniques: list[Technique]
    incidents: list[Incident]

    @cached_property
    def lookup(self) -> dict[str, dict[str, BaseModel]]:
        return {
            "reference": {r.id: r for r in self.references},
            "category": {c.id: c for c in self.categories},
            "heuristic": {h.id: h for h in self.heuristics},
            "technique": {t.id: t for t in self.techniques},
            "incident": {i.id: i for i in self.incidents},
        }

    def reference(self, rid: str) -> Reference:
        return self.lookup["reference"][rid]  # type: ignore[return-value]

    def category(self, cid: str) -> Category:
        return self.lookup["category"][cid]  # type: ignore[return-value]

    def heuristic(self, hid: str) -> Heuristic:
        return self.lookup["heuristic"][hid]  # type: ignore[return-value]

    def technique(self, tid: str) -> Technique:
        return self.lookup["technique"][tid]  # type: ignore[return-value]
```

Note: `cached_property` on a frozen pydantic model requires `ignored_types=(cached_property,)` (set above). The property stores its value in the instance `__dict__`, which frozen models still allow.

`backend/btcintel/catalog/loader.py`:
```python
"""Load the catalog from YAML and cross-check it against itself and docs/kb."""

from __future__ import annotations

import importlib
from pathlib import Path

import yaml
from pydantic import BaseModel, ValidationError

from btcintel.catalog.schema import Catalog, Category, Heuristic, Incident, Reference, Technique
from btcintel.chain.validate import is_address, is_txid

LIST_FILES: dict[str, tuple[str, type[BaseModel]]] = {
    "references": ("references.yaml", Reference),
    "categories": ("categories.yaml", Category),
    "heuristics": ("heuristics.yaml", Heuristic),
    "techniques": ("techniques.yaml", Technique),
}


class CatalogError(Exception):
    """A catalog file cannot be parsed or fails schema validation."""


def _read_yaml(path: Path) -> object:
    try:
        return yaml.safe_load(path.read_text(encoding="utf-8"))
    except FileNotFoundError as e:
        raise CatalogError(f"{path}: file not found") from e
    except yaml.YAMLError as e:
        raise CatalogError(f"{path}: invalid YAML: {e}") from e


def _validate(model: type[BaseModel], entry: object, where: str) -> BaseModel:
    label = entry.get("id", "?") if isinstance(entry, dict) else "?"
    try:
        return model.model_validate(entry)
    except ValidationError as e:
        raise CatalogError(f"{where} ({label}): {e}") from e


def load_catalog(catalog_dir: Path) -> Catalog:
    parsed: dict[str, list[BaseModel]] = {}
    for key, (filename, model) in LIST_FILES.items():
        path = catalog_dir / filename
        data = _read_yaml(path)
        if data is None:
            data = []
        if not isinstance(data, list):
            raise CatalogError(f"{path}: expected a YAML list of entries, got {type(data).__name__}")
        parsed[key] = [_validate(model, e, f"{filename} entry #{i}") for i, e in enumerate(data)]
    incidents = []
    for path in sorted((catalog_dir / "incidents").glob("*.yaml")):
        data = _read_yaml(path)
        if not isinstance(data, dict):
            raise CatalogError(f"{path}: expected one incident mapping per file")
        incidents.append(_validate(Incident, data, f"incidents/{path.name}"))
    return Catalog(**parsed, incidents=incidents)


def _resolve(dotted: str) -> str | None:
    module_name, _, attr = dotted.rpartition(".")
    try:
        module = importlib.import_module(module_name)
    except ImportError as e:
        return f"cannot import {module_name}: {e}"
    return None if hasattr(module, attr) else f"{module_name} has no attribute {attr}"


def _check_kb_page(entry_id: str, rel: str, refs: list[str], kb_dir: Path) -> list[str]:
    root = kb_dir.resolve()
    page = (kb_dir / rel).resolve()
    if not page.is_relative_to(root):
        return [f"{entry_id}: kb path {rel!r} escapes docs/kb"]
    if not page.is_file():
        return [f"{entry_id}: kb page {rel} does not exist"]
    text = page.read_text(encoding="utf-8")
    problems = []
    if not text.startswith(f"# {entry_id} "):
        problems.append(f"{entry_id}: {rel} must start with '# {entry_id} — <name>'")
    if "## References" not in text:
        problems.append(f"{entry_id}: {rel} has no '## References' section")
    problems += [f"{entry_id}: {rel} does not cite [{r}]" for r in refs if f"[{r}]" not in text]
    return problems


def check_catalog(cat: Catalog, kb_dir: Path) -> list[str]:
    """Return human-readable consistency problems (empty list = consistent)."""
    problems: list[str] = []
    ref_ids = {r.id for r in cat.references}
    heuristics = {h.id: h for h in cat.heuristics}
    techniques = {t.id: t for t in cat.techniques}

    seen: dict[str, str] = {}
    groups = (
        ("reference", cat.references), ("category", cat.categories),
        ("heuristic", cat.heuristics), ("technique", cat.techniques), ("incident", cat.incidents),
    )
    for kind, entries in groups:
        for e in entries:
            if e.id in seen:
                problems.append(f"duplicate id {e.id} ({seen[e.id]} and {kind})")
            seen[e.id] = kind

    concept_owner: dict[str, str] = {}
    for c in cat.categories:
        for concept in c.gs_concepts:
            if concept in concept_owner:
                problems.append(
                    f"GraphSense concept {concept!r} mapped by both {concept_owner[concept]} and {c.id}"
                )
            concept_owner[concept] = c.id

    def need_refs(owner: str, refs: list[str]) -> None:
        if not refs:
            problems.append(f"{owner}: at least one reference is required")
        problems.extend(f"{owner}: unknown reference {r!r}" for r in refs if r not in ref_ids)

    for h in cat.heuristics:
        need_refs(h.id, h.refs)
        if h.parent is not None:
            parent = heuristics.get(h.parent)
            if parent is None:
                problems.append(f"{h.id}: unknown parent {h.parent}")
            elif parent.parent is not None:
                problems.append(f"{h.id}: parent {h.parent} is itself a child heuristic")
        if h.implemented_by:
            err = _resolve(h.implemented_by)
            if err:
                problems.append(f"{h.id}: implemented_by {h.implemented_by!r}: {err}")

    for t in cat.techniques:
        need_refs(t.id, t.refs)
        problems.extend(
            f"{t.id}: unknown heuristic {hid}" for hid in [*t.evades, *t.detected_by]
            if hid not in heuristics
        )

    for inc in cat.incidents:
        need_refs(inc.id, inc.refs)
        for tid in [*inc.vector, *inc.techniques]:
            if tid not in techniques:
                problems.append(f"{inc.id}: unknown technique {tid}")
        for tid in inc.vector:
            if tid in techniques and techniques[tid].tactic != "acquisition":
                problems.append(f"{inc.id}: vector {tid} is not an acquisition technique")
        for s in inc.seeds:
            if s.ref not in ref_ids:
                problems.append(f"{inc.id}: seed {s.value} cites unknown reference {s.ref!r}")
            valid = is_address(s.value) if s.kind == "address" else is_txid(s.value)
            if not valid:
                problems.append(f"{inc.id}: invalid {s.kind} seed {s.value!r}")

    for entry in [*cat.heuristics, *cat.techniques, *cat.incidents]:
        problems.extend(_check_kb_page(entry.id, entry.kb, entry.refs, kb_dir))
    return problems
```

`backend/btcintel/catalog/online.py`:
```python
"""Network checks for the catalog: reference URLs resolve, incident seeds exist on chain."""

from __future__ import annotations

import time
import urllib.error
import urllib.request
from collections.abc import Iterable

from btcintel.catalog.schema import Catalog

_UA = {"User-Agent": "btc-intel-catalog-check/0.1 (+https://github.com)"}


def _status(url: str, timeout: float) -> int | str:
    req = urllib.request.Request(url, headers=_UA, method="GET")
    try:
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            return resp.status
    except urllib.error.HTTPError as e:
        return e.code
    except (urllib.error.URLError, TimeoutError, OSError) as e:
        return f"unreachable: {e}"


def check_urls(urls: Iterable[str], timeout: float = 20.0) -> list[tuple[str, str]]:
    problems = []
    for url in urls:
        status = _status(url, timeout)
        if isinstance(status, str) or status >= 400:
            problems.append((url, status if isinstance(status, str) else f"HTTP {status}"))
    return problems


def verify_seeds(
    cat: Catalog, base_url: str = "https://mempool.space/api", delay: float = 1.0
) -> list[tuple[str, str]]:
    """Every seed address/txid must be known to an Esplora API (1 request per `delay` seconds)."""
    problems = []
    for inc in cat.incidents:
        for seed in inc.seeds:
            path = f"/address/{seed.value}" if seed.kind == "address" else f"/tx/{seed.value}"
            status = _status(base_url + path, timeout=20.0)
            if status != 200:
                problems.append((f"{inc.id} {seed.kind} {seed.value}", str(status)))
            time.sleep(delay)
    return problems
```

`backend/btcintel/catalog/render.py`:
```python
"""Render docs/kb/index.md and docs/kb/ttp-matrix.md from the catalog (deterministic)."""

from __future__ import annotations

from collections import defaultdict
from pathlib import Path

from btcintel.catalog.schema import Catalog

HEADER = (
    "<!-- Generated by `btcintel catalog render` from catalog/*.yaml. Do not edit by hand. -->\n\n"
)
TACTIC_ORDER = ["acquisition", "layering", "obfuscation", "chain_hopping", "offchain", "integration"]
TACTIC_TITLES = {
    "acquisition": "Acquisition (how illicit funds are obtained)",
    "layering": "Layering",
    "obfuscation": "Obfuscation",
    "chain_hopping": "Chain hopping",
    "offchain": "Off-chain / layer 2",
    "integration": "Integration (cash-out)",
}


def _cell(text: str) -> str:
    return text.replace("|", "\\|").replace("\n", " ").strip()


def _link(entry_id: str, kb: str) -> str:
    return f"[{entry_id}]({kb})"


def _join(items: list[str]) -> str:
    return ", ".join(items) if items else "—"


def render_index(cat: Catalog) -> str:
    out = [
        HEADER + "# btc-intel knowledge base",
        "",
        "Analysis techniques (`H-*`), obfuscation / laundering / theft techniques (`O-*`) and "
        "incident case studies (`I-*`). See also the [TTP matrix](ttp-matrix.md).",
        "",
        "## Analysis techniques",
        "",
        "| ID | Name | Family | Status | Confidence |",
        "|---|---|---|---|---|",
    ]
    for h in sorted(cat.heuristics, key=lambda h: h.id):
        name = h.name if h.parent is None else f"↳ {h.name}"
        out.append(
            f"| {_link(h.id, h.kb)} | {_cell(name)} | {h.family} | {h.status} | {h.confidence} |"
        )
    out += ["", "## Obfuscation, laundering and theft techniques", "",
            "| ID | Name | Tactic | Detected by |", "|---|---|---|---|"]
    for t in sorted(cat.techniques, key=lambda t: (TACTIC_ORDER.index(t.tactic), t.id)):
        out.append(f"| {_link(t.id, t.kb)} | {_cell(t.name)} | {t.tactic} | {_join(t.detected_by)} |")
    out += ["", "## Incidents", "", "| ID | Date | Name | Loss | Attribution |", "|---|---|---|---|---|"]
    for i in sorted(cat.incidents, key=lambda i: (i.date, i.id)):
        out.append(
            f"| {_link(i.id, i.kb)} | {i.date} | {_cell(i.name)} | {_cell(i.loss)} | {i.attribution} |"
        )
    out += ["", "## Entity and abuse categories", "",
            "| ID | Kind | Risk | Stops trace | GraphSense concepts |", "|---|---|---|---|---|"]
    for c in sorted(cat.categories, key=lambda c: (c.kind, c.id)):
        out.append(
            f"| {c.id} | {c.kind} | {c.risk} | {'yes' if c.stop_trace else 'no'} | "
            f"{_join(sorted(c.gs_concepts))} |"
        )
    out += ["", "## References", ""]
    for r in sorted(cat.references, key=lambda r: r.id):
        authors = ", ".join(r.authors) if r.authors else "—"
        venue = f" {_cell(r.venue)}." if r.venue else ""
        out.append(f"- **[{r.id}]** {_cell(authors)} ({r.year}). *{_cell(r.title)}*.{venue} <{r.url}>")
    return "\n".join(out) + "\n"


def render_ttp_matrix(cat: Catalog) -> str:
    seen_in: dict[str, list[str]] = defaultdict(list)
    for inc in sorted(cat.incidents, key=lambda i: i.id):
        for tid in dict.fromkeys([*inc.vector, *inc.techniques]):
            seen_in[tid].append(inc.id)
    out = [
        HEADER + "# TTP matrix",
        "",
        "Techniques grouped by laundering-lifecycle tactic, the btc-intel analysis techniques that "
        "detect them, the analysis techniques they evade, and incidents where they were observed.",
        "",
    ]
    for tactic in TACTIC_ORDER:
        techs = sorted((t for t in cat.techniques if t.tactic == tactic), key=lambda t: t.id)
        if not techs:
            continue
        out += [f"## {TACTIC_TITLES[tactic]}", "",
                "| Technique | Detected by | Evades | Seen in | OAK |", "|---|---|---|---|---|"]
        for t in techs:
            out.append(
                f"| {_link(t.id, t.kb)} {_cell(t.name)} | {_join(t.detected_by)} | "
                f"{_join(t.evades)} | {_join(seen_in[t.id])} | {_join(t.oak)} |"
            )
        out.append("")
    out += ["## Incident playbooks", ""]
    for inc in sorted(cat.incidents, key=lambda i: (i.date, i.id)):
        chain = " → ".join(dict.fromkeys([*inc.vector, *inc.techniques]))
        out.append(f"- **{inc.id}** {_cell(inc.name)} ({inc.date}): {chain}")
    return "\n".join(out) + "\n"


def render_all(cat: Catalog) -> dict[str, str]:
    return {"index.md": render_index(cat), "ttp-matrix.md": render_ttp_matrix(cat)}


def write_all(cat: Catalog, kb_dir: Path) -> list[Path]:
    written = []
    for name, text in render_all(cat).items():
        path = kb_dir / name
        path.write_text(text, encoding="utf-8")
        written.append(path)
    return written
```
